// 【修正版】1回目から確実に成功するコピー処理 function copyShareText() { if (!latestDetailData || !latestDetailData.event) return; const ev = latestDetailData.event; const attendees = latestDetailData.attendees || []; const attendList = attendees.filter(a => a.status === '参加'); const absentList = attendees.filter(a => a.status === '不参加'); const isLog = ev.isFinished || ev.isClosed; let text = `【${ev.title}${isLog ? '（確定・結果）' : ''}】\n`; text += `📅 日時：${ev.date} ${ev.time}\n`; if (ev.description && ev.description !== '詳細情報なし') { text += `📍 場所：${ev.description}\n`; } if (!ev.isFinished) { text += `⏳ 締切：${ev.deadlineStr}\n`; } text += `\n`; text += `◯ 参加：${attendList.length}名\n`; text += `✕ 不参加：${absentList.length}名\n\n`; text += `▼ 出欠の回答・確認はこちら\n`; text += `https://liff.line.me/${LIFF_ID.trim()}?eventId=${encodeURIComponent(ev.id)}`; // ボタンのUIアニメーション用関数 const showSuccessUI = () => { const btn = document.getElementById('btnCopyShare'); const icon = document.getElementById('copyIcon'); const label = document.getElementById('copyBtnText'); icon.textContent = 'check'; label.textContent = 'コピーしました！'; btn.style.backgroundColor = 'var(--md-sys-color-primary-container)'; btn.style.color = 'var(--md-sys-color-on-primary-container)'; setTimeout(() => { icon.textContent = 'content_copy'; label.textContent = 'LINE共有用に状況をコピー'; btn.style.backgroundColor = ''; btn.style.color = ''; }, 2000); }; // フォールバック関数（WebViewのフォーカス問題に強い古典的コピー） const fallbackCopy = (str) => { const textArea = document.createElement("textarea"); textArea.value = str; textArea.style.position = "fixed"; textArea.style.left = "-9999px"; textArea.style.top = "0"; document.body.appendChild(textArea); textArea.focus(); textArea.select(); try { const successful = document.execCommand('copy'); document.body.removeChild(textArea); if (successful) { showSuccessUI(); } else { alert('コピーに失敗しました。'); } } catch (err) { document.body.removeChild(textArea); alert('コピーに失敗しました: ' + err); } }; // 画面に強制フォーカスを当ててからコピー実行 window.focus(); if (navigator.clipboard && window.isSecureContext) { navigator.clipboard.writeText(text) .then(showSuccessUI) .catch(() => { // 1回目のフォーカス外れ等で失敗した場合は、即座にフォールバックを実行 fallbackCopy(text); }); } else { fallbackCopy(text); } }
